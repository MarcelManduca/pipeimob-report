
create or replace function public.get_validation_sales_reconciliation(
  p_start date,
  p_end date
)
returns jsonb
language plpgsql
stable
security definer
set search_path = ''
as $$
declare
  result jsonb;
begin
  if p_start is null or p_end is null then
    raise exception 'start and end dates are required';
  end if;
  if p_start > p_end then
    raise exception 'start date cannot be after end date';
  end if;
  if (p_end - p_start) > 366 then
    raise exception 'date range cannot exceed 366 days';
  end if;

  with filtered as (
    select
      cs.*,
      ps.contract_code,
      vg.stage_name,
      case
        when cs.commercial_broker_name_snapshot is not null
         and cs.fiscal_broker_name_snapshot is not null
        then lower(btrim(cs.commercial_broker_name_snapshot))
          <> lower(btrim(cs.fiscal_broker_name_snapshot))
        else null
      end as broker_roles_differ
    from validation.consolidated_sales cs
    left join validation.pipeimob_sales ps
      on ps.transaction_id = cs.pipeimob_transaction_id
    left join validation.vista_gains vg
      on vg.deal_id = cs.vista_deal_id
    where (
      cs.official_sale_date between p_start and p_end
      or (
        cs.official_sale_date is null
        and cs.vista_gain_date between p_start and p_end
      )
    )
  ),
  summary as (
    select jsonb_build_object(
      'official_sales', count(*) filter (
        where pipeimob_transaction_id is not null
      ),
      'official_vgv', coalesce(sum(official_value) filter (
        where pipeimob_transaction_id is not null
      ), 0)::text,
      'matched', count(*) filter (
        where reconciliation_status = 'CONCILIADO'
      ),
      'pipeimob_without_vista_gain', count(*) filter (
        where reconciliation_status = 'PIPEIMOB_SEM_GANHO_VISTA'
      ),
      'vista_without_pipeimob_contract', count(*) filter (
        where reconciliation_status = 'VISTA_SEM_CONTRATO_PIPEIMOB'
      ),
      'value_mismatches', count(*) filter (
        where issues @> array['DIVERGENCIA_VALOR']::text[]
      ),
      'date_mismatches', count(*) filter (
        where issues @> array['DIVERGENCIA_DATA']::text[]
      ),
      'no_automatic_link', count(*) filter (
        where reconciliation_status = 'SEM_VINCULO_AUTOMATICO'
      ),
      'source_data_incomplete', count(*) filter (
        where reconciliation_status = 'DADO_FONTE_INCOMPLETO'
      ),
      'missing_commercial_broker', count(*) filter (
        where vista_deal_id is not null
          and commercial_broker_name_snapshot is null
      )
    ) as value
    from filtered
  ),
  items as (
    select coalesce(
      jsonb_agg(
        jsonb_strip_nulls(
          jsonb_build_object(
            'status', reconciliation_status,
            'issues', to_jsonb(issues),
            'pipeimob_transaction_id', pipeimob_transaction_id,
            'pipeimob_contract_code', contract_code,
            'vista_deal_id', vista_deal_id,
            'property_code', property_code,
            'official_sale_date', official_sale_date,
            'vista_gain_date', vista_gain_date,
            'delay_days', case
              when delay_hours is null then null
              else delay_hours / 24
            end,
            'official_value', official_value::text,
            'vista_value', vista_value::text,
            'value_difference', value_difference::text,
            'commercial_broker', commercial_broker_name_snapshot,
            'fiscal_broker', fiscal_broker_name_snapshot,
            'broker_roles_differ', broker_roles_differ,
            'vista_stage', stage_name
          )
        )
        order by official_sale_date, pipeimob_transaction_id, vista_deal_id
      ),
      '[]'::jsonb
    ) as value
    from filtered
  )
  select jsonb_build_object(
    'contract_version', '1.0',
    'official_source', 'pipeimob_contratos_assinados',
    'commercial_source', 'vista_status_ganho',
    'period', jsonb_build_object(
      'start', p_start,
      'end', p_end
    ),
    'generated_at', to_char(
      now() at time zone 'UTC',
      'YYYY-MM-DD"T"HH24:MI:SS.MS"Z"'
    ),
    'summary', summary.value,
    'items', items.value,
    'commercial_broker_limitation',
      'Corretor comercial do Vista; corretor fiscal do Pipeimob preservado separadamente.'
  )
  into result
  from summary cross join items;

  return result;
end;
$$;

revoke all on function public.get_validation_sales_reconciliation(date,date)
  from public, anon, authenticated;
grant execute on function public.get_validation_sales_reconciliation(date,date)
  to service_role;

comment on function public.get_validation_sales_reconciliation(date,date) is
  'Private read-only projection for the directors homologation Edge Function. Pipeimob is official; Vista Status=Ganho supplies commercial ownership.';
